/* Admin content comes only from authenticated same-origin API requests. */
(()=>{'use strict';
const $=id=>document.getElementById(id),labels={new:'新回報',read:'已讀',planned:'規劃中',done:'已完成',closed:'已結案'};let view='feedback',page=0,loading=false;
function el(tag,text,cls){const e=document.createElement(tag);if(text!==undefined)e.textContent=text;if(cls)e.className=cls;return e;}
function time(n){return new Date(n*1000).toLocaleString('zh-TW',{timeZone:'Asia/Taipei',hour12:false});}
function duration(ms){if(ms===null||ms===undefined)return '尚無紀錄';const seconds=Math.floor(ms/1000),h=Math.floor(seconds/3600),m=Math.floor(seconds%3600/60),s=seconds%60;return (h?h+' 小時 ':'')+(m?m+' 分 ':'')+s+' 秒';}
function money(c){return '$'+(c/100).toLocaleString('zh-TW',{maximumFractionDigits:2});}
async function api(path,data){const response=await fetch('/admin/api/'+path,{method:data?'POST':'GET',credentials:'same-origin',headers:data?{'Content-Type':'application/json','X-Admin-Action':'1'}:{},...(data?{body:JSON.stringify(data)}:{})});let body;try{body=await response.json();}catch{throw Error('登入可能已過期，請重新整理並登入。');}if(!response.ok){const e=Error(body.error?.message||'暫時無法取得資料。');e.code=body.error?.code;if(e.code==='ADMIN_LOGIN_REQUIRED')showLogin();throw e;}return body;}
function message(t,error=false){$('message').textContent=t;$('message').className=error?'error':'';}
async function overview(){const d=await api('overview');$('identity').textContent='登入：'+d.actor+' · 所有時間為台北時間';$('overview').replaceChildren(...[['未讀回報',d.unread],['全部回報',d.feedback],['有效成績',d.runs],['匿名玩家',d.players]].map(([label,n])=>{const a=el('article');a.append(el('small',label),el('b',String(n)));return a;}));}
function report(r){const box=el('details',undefined,'report'),summary=el('summary'),meta=el('div',undefined,'meta');meta.append(el('span',r.kind==='wish'?'✧ 許願':'⚑ 問題回報','badge'),el('span',labels[r.status],'badge'),el('time',time(r.created_at)));summary.append(meta,el('h2',r.title));box.append(summary,el('p',r.body,'body'));if(r.steps)box.append(el('div','重現步驟\n'+r.steps,'steps'));box.append(el('p','版本 '+r.version+(r.game_date?' · '+r.game_date+' · '+(r.mode==='life'?'人生模式':'自由模式'):'')+' · 編號 '+r.id));
 const form=el('form',undefined,'edit'),label=el('label','處理狀態'),select=el('select');for(const [value,text]of Object.entries(labels)){const o=el('option',text);o.value=value;o.selected=value===r.status;select.append(o);}label.append(select);const notes=el('label','管理筆記（只有後台可見）'),area=el('textarea');area.maxLength=2000;area.value=r.admin_note;notes.append(area);const save=el('button','儲存處理紀錄');save.type='submit';form.append(label,notes,save);form.onsubmit=async e=>{e.preventDefault();save.disabled=true;try{await api('feedback/'+r.id,{status:select.value,note:area.value});message('已儲存處理紀錄。');await overview();r.status=select.value;meta.children[1].textContent=labels[r.status];}catch(err){message(err.message,true);}finally{save.disabled=false;}};box.append(form);return box;}
function run(r){const box=el('article',undefined,'run');box.append(el('div',(r.mode==='life'?'人生模式':'自由模式')+' · '+(r.financing?'融資榜':'現股榜')+' · '+time(r.submitted_at),'meta'),el('h2',r.nickname+' · '+(r.outcome==='won'?'成功超越 0050':'未能超越 0050')),el('div',(r.score>=0?'+':'')+(r.score/100).toFixed(2)+' 百分點','score'),el('p','總資產 '+money(r.equity_cents)+' · 最大回撤 '+(r.drawdown_bps/100).toFixed(2)+'% · 成交 '+r.trades+' 筆'),el('p','編號 '+r.id));box.append(el('p','遊玩時間 '+duration(r.play_time_ms)+(r.play_time_partial?'（從新版開始累計）':''),'play-stats'));const uses=r.skill_uses?JSON.parse(r.skill_uses):null;box.append(el('p',uses?'技能使用：媽媽10塊 '+uses.mother+' 次 · 林北VIP拉 '+uses.vip+' 次 · 奇門遁甲 '+uses.divination+' 次':'技能使用：尚無紀錄','skill-stats'));const margin=JSON.parse(r.margin_summary||'{}');if(r.financing)box.append(el('p','累計融資動用 '+money(margin.totalBorrowedCents||0)+' · 利息 '+money(margin.totalInterestCents||0)+' · 斷頭 '+(margin.liquidations||0)+' 次'));if(r.hidden){box.append(el('p','已下架：'+r.moderation_note));return box;}const button=el('button','下架異常成績','hide');button.onclick=()=>{button.hidden=true;const form=el('form',undefined,'edit'),label=el('label','下架原因（保留紀錄，不刪除）'),area=el('textarea');area.required=true;area.minLength=2;area.maxLength=500;label.append(area);const save=el('button','確認下架');save.type='submit';const cancel=el('button','取消');cancel.type='button';cancel.onclick=()=>{form.remove();button.hidden=false;};form.append(label,cancel,save);form.onsubmit=async e=>{e.preventDefault();save.disabled=true;try{await api('runs/'+r.id+'/hide',{note:area.value});message('成績已下架，該玩家排行榜會重新選取剩餘最佳成績。');await load();}catch(err){message(err.message,true);save.disabled=false;}};box.append(form);};box.append(button);return box;}

function announcementEditor(a){const form=el('form',undefined,'announcement-editor edit'),titleLabel=el('label','公告標題'),title=el('input'),bodyLabel=el('label','公告內容'),body=el('textarea');title.maxLength=60;title.required=true;title.value=a?.title||'';body.maxLength=2000;body.rows=8;body.required=true;body.value=a?.body||'';titleLabel.append(title);bodyLabel.append(body);form.append(el('h2','發布一則公告'),el('p','更新後會取代目前公告。玩家依發布時間顯示一次新公告，首頁跑馬燈與「更新及公告」也會同步更新。'),el('p',a?'目前發布時間：'+new Date(a.publishedAt).toLocaleString('zh-TW',{timeZone:'Asia/Taipei',hour12:false}):'尚未發布公告。'),titleLabel,bodyLabel);const submit=el('button',a?'更新並發布公告':'發布公告');submit.type='submit';form.append(submit);form.onsubmit=async e=>{e.preventDefault();submit.disabled=true;try{const r=await api('announcement',{title:title.value,body:body.value});$('content').replaceChildren(announcementEditor(r.announcement));message('公告已發布；玩家回到起始畫面時會看到。');}catch(err){message(err.message,true);submit.disabled=false;}};return form;}

async function load(){if(loading)return;loading=true;$('refresh').disabled=true;message('讀取中…');try{await overview();if(view==='announcement'){const d=await api('announcement');$('content').replaceChildren(announcementEditor(d.announcement));$('prev').disabled=$('next').disabled=true;$('pageInfo').textContent='';message('');return;}const path=view==='feedback'?'feedback?status='+$('statusFilter').value+'&page='+page:'runs?page='+page,d=await api(path);$('content').replaceChildren(...(d.items.length?d.items.map(view==='feedback'?report:run):[el('div',view==='feedback'?'目前沒有這個狀態的回報。':'目前沒有提交紀錄。','empty')]));$('prev').disabled=page===0;$('next').disabled=!d.hasMore;$('pageInfo').textContent='第 '+(page+1)+' 頁';message('');}catch(e){message(e.message,true);}finally{loading=false;$('refresh').disabled=false;}}
function tab(next){view=next;page=0;$('feedbackTab').setAttribute('aria-pressed',view==='feedback');$('runsTab').setAttribute('aria-pressed',view==='runs');$('announcementTab').setAttribute('aria-pressed',view==='announcement');document.querySelector('.paging').hidden=view==='announcement';$('statusFilterLabel').hidden=view!=='feedback';load();}
$('announcementTab').onclick=()=>tab('announcement');$('feedbackTab').onclick=()=>tab('feedback');$('runsTab').onclick=()=>tab('runs');$('statusFilter').onchange=()=>{page=0;load();};$('prev').onclick=()=>{page=Math.max(0,page-1);load();};$('next').onclick=()=>{page++;load();};$('refresh').onclick=load;
function showLogin(msg=''){$('loginPanel').hidden=false;$('adminPanel').hidden=true;$('logout').hidden=true;$('content').replaceChildren();$('loginMessage').textContent=msg;}
function showAdmin(){$('loginPanel').hidden=true;$('adminPanel').hidden=false;$('logout').hidden=false;page=0;load();}
$('loginForm').onsubmit=async e=>{e.preventDefault();const button=$('loginForm').querySelector('button');button.disabled=true;$('loginMessage').textContent='登入中…';try{await api('login',{username:$('adminUsername').value.trim(),password:$('adminPassword').value});$('adminPassword').value='';showAdmin();}catch(err){$('loginMessage').textContent=err.message;}finally{button.disabled=false;}};
$('logout').onclick=async()=>{try{await api('logout',{});showLogin('已登出。');}catch(e){message(e.message,true);}};
api('me').then(showAdmin).catch(()=>showLogin());
})();
